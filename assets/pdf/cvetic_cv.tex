% Section opener: top-aligned parboxes (label | first block).
% \strut on the label gives the same first-line box height as normal paragraphs (and bib entries),
% so the cap line lines up with body text; without it, \textsc can sit optically high vs. prose.
\newcommand{\cvsection}[2]{%
  \noalign{\vspace{\cvsectionabove}}%
  \parbox[t]{\cvlabelwidth}{\raggedright\strut\textsc{#1}} &

}

  }%
}%
